type Small is array [4] integer
type Big is array [6] integer

routine findMax(arr: array integer) : integer is
    var maxVal is arr[0]

    for i in 1 .. arr.size - 1 loop
        if arr[i] > maxVal then
            maxVal := arr[i]
        end
    end

    return maxVal
end

routine main() is
    var a: Small
    a[0] := 15; a[1] := 42; a[2] := 8; a[3] := 23

    var b: Big
    b[0] := 3; b[1] := 99; b[2] := 12
    b[3] := 7; b[4] := 55; b[5] := 1

    print "Max of 4 elements is: ", findMax(a)
    print "Max of 6 elements is: ", findMax(b)
end
