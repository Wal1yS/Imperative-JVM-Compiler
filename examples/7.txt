type IntArray is array [5] integer

routine main() is
    var arr: IntArray
    var temp: integer
    var i: integer
    var n is 5
    var swapped: boolean is true

    // comment
    arr[0] := 64; arr[1] := 34; arr[2] := 25; arr[3] := 12; arr[4] := 22

    while swapped = true loop
        swapped := false
        i := 0

        while i < n - 1 loop
            if arr[i] > arr[i + 1] then
                temp := arr[i]
                arr[i] := arr[i + 1]
                arr[i + 1] := temp
                swapped := true
            end
            i := i + 1
        end
        n := n - 1
    end

    print "Sorted array smallest element is: ", arr[0]
    print "Sorted array largest element is: ", arr[4]
end
